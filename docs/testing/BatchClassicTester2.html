Модуль для массового тестирования качества классических методов сегментации относительно Ground Truth.

Предназначен для автоматизированной оценки эффективности алгоритмов пороговой обработки и выделения границ (OpenCV, Scikit-learn, PyTorch) на датасете с размеченными данными (ADE20K). В отличие от BatchClassicTester, данный модуль сравнивает предсказания с эталонными масками, а не между реализациями.

Основные возможности:

  • Загрузка изображений и GT-масок с автоматическим ресайзом и бинаризацией многоклассовых аннотаций
  • Последовательное тестирование 21 метода из трёх бэкендов с единым интерфейсом
  • Расчёт агрегированных метрик: IoU, Dice, Precision, Recall, F1, Accuracy, MAE, Hausdorff
  • Продвинутый трекер прогресса с ETA, скоростью выполнения и статистикой ошибок
  • Устойчивость к прерываниям: resume после Ctrl+C/kill с автосохранением прогресса
  • Экспорт отчётов в CSV, JSON, Markdown и автоматическая генерация аналитических графиков
  • Управление памятью: очистка GPU-кэша и сборка мусора между итерациями

Примечание:

  • Многоклассовые маски ADE20K автоматически конвертируются в бинарные: наиболее частый класс = фон (0), остальные = объект (255).
  • Для валидации консистентности реализаций между библиотеками используйте BatchClassicTester.

196    def _update_progress_bar(
197        self, pbar, current_count: int, method_name: str, img_name: str
198    ) -> None:
199        """Обновляет postfix прогресс-бара динамической статистикой.
200
201        Рассчитывает прошедшее время, ETA, скорость обработки (тестов/мин),
202        а также общую долю ошибок на текущий момент.
203
204        Args:
205            pbar: Экземпляр `tqdm` для обновления.
206            current_count: Количество уже выполненных тестов.
207            method_name: Имя текущего метода сегментации.
208            img_name: Имя текущего изображения.
209        """
210        elapsed: float = time.time() - self._start_time
211        rate: float = current_count / elapsed if elapsed > 0 else 0
212        remaining: float = (self._total_tests - current_count) / rate if rate > 0 else 0
213
214        # Форматирование времени
215        def fmt_time(seconds: float) -> str:
216            if seconds < 60:
217                return f"{seconds:.0f}с"
218            elif seconds < 3600:
219                return f"{seconds / 60:.1f}м"
220            else:
221                return f"{seconds / 3600:.1f}ч"
222
223        total_errors: int = sum(len(errs) for errs in self.errors.values())
224        error_rate: float = total_errors / current_count if current_count > 0 else 0
225
226        pbar.set_postfix(
227            {
228                "img": img_name[:15],
229                "method": method_name.split("_")[0],
230                "elapsed": fmt_time(elapsed),
231                "eta": fmt_time(remaining),
232                "rate": f"{rate * 60:.1f}/мин",
233                "errors": f"{total_errors}({error_rate * 100:.1f}%)",
234            }
235        )

238    def _save_progress(self, force: bool = False) -> None:
239        """Сохраняет текущий прогресс и промежуточные результаты на диск.

241        Использует атомарную перезапись CSV для предотвращения повреждения данных

243        если `force=False`.
244
245        Args:
246            force: Если `True`, сохраняет прогресс немедленно, игнорируя таймер.
247        """
248        now: float = time.time()
249        # Сохраняем если прошло достаточно времени или по принуждению
250        if not force and (now - self._last_save_time) < 30:  # не чаще 30 сек
251            return
252
253        try:
254            # 1. Сохраняем метаданные прогресса
255            progress: Dict[str, Any] = {
256                "processed_count": self._processed_count,
257                "total_tests": self._total_tests,
258                "start_time": self._start_time,
259                "last_update": now,
260                "methods_done": list(self.results.keys()),
261            }
262            with open(self.progress_file, "w") as f:
263                json.dump(progress, f, indent=2)
264
265            # 2. Сохраняем промежуточные результаты (atomic write)
266            if self.results:
267                df_temp: pd.DataFrame = self._aggregate_results()
268                df_temp.to_csv(self.temp_results_file, index=False, float_format="%.4f")
269                # Atomic rename
270                final_path: Path = self.temp_results_file.with_suffix(".csv")
271                self.temp_results_file.replace(final_path)
272
273            self._last_save_time = now
274            print(
275                f"\n💾 Автосохранение: {self._processed_count}/{self._total_tests} ✅"
276            )
277
278        except Exception as e:
279            print(f"\n⚠️  Ошибка автосохранения: {e}")

312    def _load_images_with_masks(self) -> List[Tuple[str, np.ndarray, np.ndarray]]:
313        """Загружает пары (изображение, ground truth маска) из датасета ADE20K.
314
315        Автоматически ресайзит данные до `self.image_size`. Пропускает изображения,
316        для которых не найдены соответствующие маски.
317
318        Returns:
319            Список кортежей `(имя_файла, изображение_HxWxC_uint8, бинарная_маска_HxW_uint8)`.
320        """
321        images_dir: Path = self.ade20k_root / "images" / self.split
322        masks_dir: Path = self.ade20k_root / "annotations" / self.split
323
324        if not images_dir.exists():
325            raise FileNotFoundError(f"Images directory not found: {images_dir}")
326        if not masks_dir.exists():
327            raise FileNotFoundError(f"Masks directory not found: {masks_dir}")
328
329        # Получаем список изображений
330        image_files: List[str] = sorted(
331            [f for f in os.listdir(images_dir) if f.endswith((".jpg", ".jpeg", ".png"))]
332        )
333
334        if self.max_images:
335            image_files = image_files[: self.max_images]
336
337        print(f"📥 Загрузка {len(image_files)} изображений из {self.split}...")
338
339        data: List[Tuple[str, np.ndarray, np.ndarray]] = []
340        for img_file in tqdm(image_files, desc="Loading images"):
341            mask_file: str = img_file.rsplit(".", 1)[0] + ".png"
342            img_path: Path = images_dir / img_file
343            mask_path: Path = masks_dir / mask_file
344
345            if not mask_path.exists():
346                print(f"⚠️  Mask not found for {img_file}, skipping")
347                continue
348
349            try:
350                # Загрузка изображения
351                img: Image.Image = Image.open(img_path).convert("RGB")
352                img_array: np.ndarray = np.array(
353                    img.resize(self.image_size, Image.Resampling.BILINEAR)
354                )
355
356                # Загрузка маски
357                mask_pil: Image.Image = Image.open(mask_path)
358                mask_array: np.ndarray = np.array(
359                    mask_pil.resize(self.image_size, Image.Resampling.NEAREST)
360                )
361
362                # Конвертация многоклассовой маски в бинарную
363                # Стратегия: самый частый класс = фон (0), всё остальное = объект (255)
364                binary_mask: np.ndarray = self._multiclass_to_binary(mask_array)
365
366                data.append((img_file, img_array, binary_mask))
367
368            except Exception as e:
369                print(f"❌ Error loading {img_file}: {e}")
370                continue
371
372        print(f"✅ Загружено {len(data)} пар изображение-маска")
373        return data

376    def _multiclass_to_binary(self, mask: np.ndarray) -> np.ndarray:
377        """Конвертирует многоклассовую маску в бинарную.
378
379        Стратегия: класс с наибольшей частотой пикселей считается фоном (0),
380        все остальные классы объединяются в объект (255).
381
382        Args:
383            mask: Входная маска с целочисленными метками классов.
384
385        Returns:
386            Бинарная маска `np.uint8`, где 255 = объект, 0 = фон.
387        """
388        # Находим самый частый класс (предполагаем, что это фон)
389        unique: np.ndarray
390        counts: np.ndarray
391        unique, counts = np.unique(mask, return_counts=True)
392        background_class = unique[np.argmax(counts)]
393
394        # Бинаризация: объект = всё, что не фон
395        binary: np.ndarray = (mask != background_class).astype(np.uint8) * 255
396        return binary

633        """Агрегирует накопленные результаты в сводную таблицу.

659                    row[f"{metric_name}_max"] = np.max(values)
660                else:
661                    row[f"{metric_name}_mean"] = np.nan
662
663            # Время выполнения
664            times: List[float] = self.execution_times.get(method_name, [])
665            if times:

667                row["time_std_s"] = np.std(times)
668            else:
669                row["time_mean_s"] = np.nan
670
671            # Ошибки
672            errors: List[str] = self.errors.get(method_name, [])
673            row["error_count"] = len(errors)
674            row["error_rate"] = len(errors) / max(row["Images_Tested"], 1)

678        df: pd.DataFrame = pd.DataFrame(rows)
679
680        # Сортировка по IoU

196    # ──────────────────────────────────────────────────────────────────────
197    def _update_progress_bar(
198        self, pbar, current_count: int, method_name: str, img_name: str
199    ) -> None:
200        """Обновляет postfix прогресс-бара динамической статистикой.
201
202        Рассчитывает прошедшее время, ETA, скорость обработки (тестов/мин),
203        а также общую долю ошибок на текущий момент.
204
205        Args:
206            pbar: Экземпляр `tqdm` для обновления.
207            current_count: Количество уже выполненных тестов.
208            method_name: Имя текущего метода сегментации.
209            img_name: Имя текущего изображения.
210        """
211        elapsed: float = time.time() - self._start_time
212        rate: float = current_count / elapsed if elapsed > 0 else 0
213        remaining: float = (self._total_tests - current_count) / rate if rate > 0 else 0
214
215        # Форматирование времени
216        def fmt_time(seconds: float) -> str:
217            if seconds < 60:
218                return f"{seconds:.0f}с"
219            elif seconds < 3600:
220                return f"{seconds / 60:.1f}м"
221            else:
222                return f"{seconds / 3600:.1f}ч"
223
224        total_errors: int = sum(len(errs) for errs in self.errors.values())
225        error_rate: float = total_errors / current_count if current_count > 0 else 0
226
227        pbar.set_postfix(
228            {
229                "img": img_name[:15],
230                "method": method_name.split("_")[0],
231                "elapsed": fmt_time(elapsed),
232                "eta": fmt_time(remaining),
233                "rate": f"{rate * 60:.1f}/мин",
234                "errors": f"{total_errors}({error_rate * 100:.1f}%)",
235            }
236        )

242        Использует атомарную перезапись CSV для предотвращения повреждения данных

256            progress: Dict[str, Any] = {

259                "start_time": self._start_time,

261                "methods_done": list(self.results.keys()),
262            }

264                json.dump(progress, f, indent=2)

266            # 2. Сохраняем промежуточные результаты (atomic write)
267            if self.results:
268                df_temp: pd.DataFrame = self._aggregate_results()
269                df_temp.to_csv(self.temp_results_file, index=False, float_format="%.4f")
270                # Atomic rename
271                final_path: Path = self.temp_results_file.with_suffix(".csv")
272                self.temp_results_file.replace(final_path)
273
274            self._last_save_time = now
275            print(
276                f"\n💾 Автосохранение: {self._processed_count}/{self._total_tests} ✅"
277            )
278
279        except Exception as e:
280            print(f"\n⚠️  Ошибка автосохранения: {e}")

313    def _load_images_with_masks(self) -> List[Tuple[str, np.ndarray, np.ndarray]]:
314        """Загружает пары (изображение, ground truth маска) из датасета ADE20K.
315
316        Автоматически ресайзит данные до `self.image_size`. Пропускает изображения,
317        для которых не найдены соответствующие маски.
318
319        Returns:
320            Список кортежей `(имя_файла, изображение_HxWxC_uint8, бинарная_маска_HxW_uint8)`.
321        """
322        images_dir: Path = self.ade20k_root / "images" / self.split
323        masks_dir: Path = self.ade20k_root / "annotations" / self.split
324
325        if not images_dir.exists():
326            raise FileNotFoundError(f"Images directory not found: {images_dir}")
327        if not masks_dir.exists():
328            raise FileNotFoundError(f"Masks directory not found: {masks_dir}")
329
330        # Получаем список изображений
331        image_files: List[str] = sorted(
332            [f for f in os.listdir(images_dir) if f.endswith((".jpg", ".jpeg", ".png"))]
333        )
334
335        if self.max_images:
336            image_files = image_files[: self.max_images]
337
338        print(f"📥 Загрузка {len(image_files)} изображений из {self.split}...")
339
340        data: List[Tuple[str, np.ndarray, np.ndarray]] = []
341        for img_file in tqdm(image_files, desc="Loading images"):
342            mask_file: str = img_file.rsplit(".", 1)[0] + ".png"
343            img_path: Path = images_dir / img_file
344            mask_path: Path = masks_dir / mask_file
345
346            if not mask_path.exists():
347                print(f"⚠️  Mask not found for {img_file}, skipping")
348                continue
349
350            try:
351                # Загрузка изображения
352                img: Image.Image = Image.open(img_path).convert("RGB")
353                img_array: np.ndarray = np.array(
354                    img.resize(self.image_size, Image.Resampling.BILINEAR)
355                )
356
357                # Загрузка маски
358                mask_pil: Image.Image = Image.open(mask_path)
359                mask_array: np.ndarray = np.array(
360                    mask_pil.resize(self.image_size, Image.Resampling.NEAREST)
361                )
362
363                # Конвертация многоклассовой маски в бинарную
364                # Стратегия: самый частый класс = фон (0), всё остальное = объект (255)
365                binary_mask: np.ndarray = self._multiclass_to_binary(mask_array)
366
367                data.append((img_file, img_array, binary_mask))
368
369            except Exception as e:
370                print(f"❌ Error loading {img_file}: {e}")
371                continue
372
373        print(f"✅ Загружено {len(data)} пар изображение-маска")
374        return data

376    # ──────────────────────────────────────────────────────────────────────
377    def _multiclass_to_binary(self, mask: np.ndarray) -> np.ndarray:
378        """Конвертирует многоклассовую маску в бинарную.
379
380        Стратегия: класс с наибольшей частотой пикселей считается фоном (0),
381        все остальные классы объединяются в объект (255).
382
383        Args:
384            mask: Входная маска с целочисленными метками классов.
385
386        Returns:
387            Бинарная маска `np.uint8`, где 255 = объект, 0 = фон.
388        """
389        # Находим самый частый класс (предполагаем, что это фон)
390        unique: np.ndarray
391        counts: np.ndarray
392        unique, counts = np.unique(mask, return_counts=True)
393        background_class = unique[np.argmax(counts)]
394
395        # Бинаризация: объект = всё, что не фон
396        binary: np.ndarray = (mask != background_class).astype(np.uint8) * 255
397        return binary

400    def _run_single_test(
401        self, method_name: str, segmenter: Any, image: np.ndarray, gt_mask: np.ndarray
402    ) -> Tuple[Optional[Dict[str, float]], Optional[float], Optional[str]]:
403        """Выполняет один полный цикл тестирования: сегментация + расчёт метрик.
404
405        Args:
406            method_name: Идентификатор метода (для логирования).
407            segmenter: Экземпляр класса-сегментатора с методом `.segment()`.
408            image: Входное изображение `np.ndarray`.
409            gt_mask: Ground truth бинарная маска `np.ndarray`.
410
411        Returns:
412            Кортеж `(метрики_dict, время_выполнения_сек, сообщение_об_ошибке)`.
413            При успехе `error_msg` равен `None`, при ошибке первые два элемента `None`.
414        """
415        try:
416            start_time: float = time.time()
417
418            # Сегментация
419            pred_mask: np.ndarray = segmenter.segment(image)
420
421            # Ресайз предсказания к размеру GT если нужно
422            if pred_mask.shape != gt_mask.shape:
423                from skimage.transform import resize
424
425                pred_mask = resize(
426                    pred_mask, gt_mask.shape, order=0, preserve_range=True
427                ).astype(np.uint8)
428
429            exec_time: float = time.time() - start_time
430
431            # Расчёт метрик
432            metrics: Dict[str, float] = SegmentationMetrics.calculate_all_metrics(
433                pred_mask=pred_mask,
434                gt_mask=gt_mask,
435                threshold=0.5,
436                include_hausdorff=True,
437            )
438
439            return metrics, exec_time, None
440
441        except Exception as e:
442            error_msg: str = f"{type(e).__name__}: {str(e)}"
443            return None, None, error_msg

527    # ──────────────────────────────────────────────────────────────────────
528    def run_batch_test(self) -> pd.DataFrame:
529        """Запускает основной цикл массового тестирования всех методов на всех изображениях.
530
531        Обрабатывает прерывания, обновляет прогресс-бар, управляет памятью GPU/CPU
532        и автоматически сохраняет промежуточные результаты.
533
534        Returns:
535            `pd.DataFrame` с агрегированными метриками, временем выполнения и статистикой ошибок.
536        """
537        # Загрузка данных
538        test_data: List[Tuple[str, np.ndarray, np.ndarray]] = (
539            self._load_images_with_masks()
540        )
541        if not test_data:
542            raise ValueError("No test data loaded!")
543
544        # Получение методов
545        methods: Dict[str, Any] = self._get_classic_methods()
546        total_images: int = len(test_data)
547        total_methods: int = len(methods)
548
549        print(f"🔧 Тестируем {total_methods} методов на {total_images} изображениях")
550        print(f"📊 Всего тестов: {total_images * total_methods}")
551        print(f"💾 Автосохранение каждые {self.autosave_interval} изображений")
552
553        # Инициализация прогресса
554        self._init_progress_tracking(total_images, total_methods)
555
556        # Прогресс-бар с расширенными настройками
557        from tqdm import tqdm
558
559        with tqdm(
560            total=self._total_tests,
561            desc="🧪 Тестирование",
562            unit="тест",
563            leave=True,
564            dynamic_ncols=False,
565            ncols=140,  # подберите под ширину вашего терминала
566            mininterval=0.5,
567        ) as pbar:
568
569            # Пропускаем уже выполненные тесты если resume=True
570            if self.resume and self._processed_count > 0:
571                pbar.update(self._processed_count)
572                print(f"⏭️  Пропущено {self._processed_count} выполненных тестов")
573
574            # Основной цикл
575            for img_idx, (img_name, image, gt_mask) in enumerate(test_data):
576                for method_idx, (method_name, segmenter) in enumerate(methods.items()):
577
578                    # Пропуск если уже выполнено (при resume)
579                    # test_key = f"{img_name}:{method_name}"
580                    if self.resume and self._processed_count > 0:
581                        # Простая эвристика: если счётчик больше — пропускаем
582                        if self._processed_count >= (
583                            img_idx * total_methods + method_idx + 1
584                        ):
585                            continue
586
587                    # Запуск теста
588                    metrics, exec_time, error = self._run_single_test(
589                        method_name, segmenter, image, gt_mask
590                    )
591
592                    # Обработка результатов
593                    if error:
594                        self.errors[method_name].append(f"{img_name}: {error}")
595                    elif metrics:
596                        for metric_name in self.metrics_to_aggregate:
597                            if metric_name in metrics:
598                                self.results[method_name][metric_name].append(
599                                    metrics[metric_name]
600                                )
601                        if exec_time is not None:
602                            self.execution_times[method_name].append(exec_time)
603                    # Обновление прогресса
604                    self._processed_count += 1
605                    pbar.update(1)
606                    self._update_progress_bar(
607                        pbar, self._processed_count, method_name, img_name
608                    )
609
610                    # Автосохранение каждые N изображений
611                    if img_idx % self.autosave_interval == 0 and img_idx > 0:
612                        self._save_progress()
613
614                # Очистка памяти после каждого изображения
615                if torch.cuda.is_available():
616                    torch.cuda.empty_cache()
617                    torch.cuda.synchronize()
618                gc.collect()
619
620        # Финальное сохранение
621        print("\n🏁 Завершение тестирования...")
622        self._save_progress(force=True)
623
624        # Убираем временные файлы
625        if self.progress_file.exists():
626            self.progress_file.unlink()
627        if self.temp_results_file.exists():
628            self.temp_results_file.unlink()
629
630        return self._aggregate_results()
631
632    # ──────────────────────────────────────────────────────────────────────
633    def _aggregate_results(self) -> pd.DataFrame:
634        """Агрегирует накопленные результаты в сводную таблицу.
635
636        Для каждого метода рассчитывает среднее, стандартное отклонение, min и max
637        по каждой метрике, а также среднее время выполнения и долю ошибок.
638        Сортирует таблицу по убыванию `iou_mean`.
639
640        Returns:
641            `pd.DataFrame` со столбцами `Method`, `{metric}_{stat}`, `time_mean_s`,
642            `error_count`, `error_rate` и `Images_Tested`.
643        """
644        rows: List[Dict[str, Any]] = []
645
646        for method_name in self.results:
647            images_tested: int = len(self.results[method_name].get("iou", []))
648            row: Dict[str, Any] = {
649                "Method": method_name,
