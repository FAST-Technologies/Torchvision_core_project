Модульные тесты для оптимизированного PyTorch-сегментера (TorchSegmenter2).

165    @pytest.mark.parametrize("precision", ["fp32", "fp16", "bf16"])
166    def test_precision_modes(self, test_image: np.ndarray, precision: str) -> None:
167        """Проверяет работу с fp32/fp16/bf16 и корректный fallback на CPU.
168
169        На CPU fp16/bf16 должны автоматически откатываться к fp32 с предупреждением,
170        но не вызывать краш. На GPU используется нативная точность.
171        """
172        device = "cuda" if torch.cuda.is_available() and precision != "fp32" else "cpu"
173        # tolerance = 0.05 if precision in ["fp16", "bf16"] and device == "cpu" else 0.01
174        with warnings.catch_warnings(record=True) as w:
175            warnings.simplefilter("always")
176            segmenter = TorchSegmenter2(
177                method="sobel_edge",
178                device=device,
179                precision=precision,
180                use_compile=False,
181            )
182
183            mask = segmenter.segment(test_image)
184            if precision in ["fp16", "bf16"] and device == "cpu":
185                # Проверяем, что внутренний dtype — fp32 (fallback сработал)
186                assert (
187                    segmenter.dtype == torch.float32
188                ), f"Ожидался fallback на fp32 для {precision} на CPU, но получен {segmenter.dtype}"
189
190                # Опционально: проверяем, что было хоть какое-то предупреждение
191                if w:
192                    warning_texts = [str(warn.message).lower() for warn in w]
193                    assert any(
194                        "cpu" in txt or "fallback" in txt or "fp32" in txt for txt in warning_texts
195                    ), f"Ожидалось предупреждение о fallback, но получено: {warning_texts}"
196            # Для GPU проверяем, что точность сохранена
197            elif device == "cuda":
198                expected_dtype = {"fp32": torch.float32, "fp16": torch.float16, "bf16": torch.bfloat16}[precision]
199
200                if expected_dtype != torch.float16 and expected_dtype != torch.bfloat16:
201                    assert (
202                        segmenter.dtype == expected_dtype
203                    ), f"На GPU ожидался {expected_dtype}, но получен {segmenter.dtype}"
204
205        assert mask.dtype == np.uint8
206        assert mask.shape == test_image.shape[:2]
207        assert set(np.unique(mask)).issubset({0, 255})
208
209        assert any(mask.flatten() > 0) or any(mask.flatten() == 0)
